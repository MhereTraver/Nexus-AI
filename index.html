export class ResilientRedisCache<T> {
  private client: RedisCluster;
  private circuitBreaker: CircuitBreaker;

  constructor(config: CacheConfig) {
    this.client = new RedisCluster(config.nodes);
    this.circuitBreaker = new CircuitBreaker({
      failureThreshold: 3,
      resetTimeoutMs: 5000
    });
  }

  async getOrSet(key: string, fetcher: () => Promise<T>, ttlSeconds = 300): Promise<T> {
    if (!this.circuitBreaker.isOpen()) {
      const cached = await this.client.get(key);
      if (cached) return JSON.parse(cached);
    }
    
    // Fallback seamlessly to upstream source
    const fresh = await fetcher();
    this.client.setex(key, ttlSeconds, JSON.stringify(fresh)).catch(console.error);
    return fresh;
  }
}
